% CONSORT diagram: pretraining population -> first-line ascertainment -> final
% (prediction) cohort and its temporal split. House style as the other figures:
% eyebrow + heading per band, soft panels, stage boxes, accents only on the
% first and last steps. Exclusions leave the flow line between steps.
\def\cfn#1{{\fontsize{11}{13.5}\selectfont\bfseries\textit{n}\,=\,#1}}
\def\cfsub#1{{\fontsize{8.6}{10.4}\selectfont\color{OperaInkSecondary}#1}}
\begin{tikzpicture}[opera,
    every node/.append style={execute at begin node={\hyphenpenalty=10000\exhyphenpenalty=10000}},
    flowstep/.style={stage, text width=5.9cm, inner ysep=2.4mm, font=\sffamily\fontsize{10}{12.2}\selectfont},
    excl/.style={rounded corners=0.8mm, draw=OperaSienna!45, fill=white,
      line width=0.55pt, text width=5.5cm, align=left, inner xsep=3mm, inner ysep=2.6mm,
      font=\sffamily\fontsize{9}{11}\selectfont, text=OperaInkSecondary},
    exclarrow/.style={-{Latex[length=1.7mm,width=1.1mm]}, line width=0.6pt,
      draw=OperaSienna!80},
    splitbox/.style={stage, text width=3.2cm, inner ysep=2.4mm, draw=OperaTeal!65,
      font=\sffamily\fontsize{10}{12.2}\selectfont,
      line width=0.7pt},
    band/.style={softpanel, rounded corners=1.6mm, inner xsep=4.5mm, inner ysep=3.4mm}]

  % Steps
  \node[flowstep, stageblue] (pre)
    {DALY-CARE patients with a lymphoid cancer or precursor diagnosis\\
     \cfn{78,668}\\\cfsub{histories before 2022 used for pretraining}};
  \node[flowstep, below=44mm of pre] (index)
    {First-line treatment index identified\\\cfn{38,635}};
  \node[flowstep, below=13mm of index] (classified)
    {Diagnosis assigned\\\cfn{37,566}};
  \node[flowstep, below=13mm of classified] (firstline)
    {First-line cohort\\\cfn{37,557}};
  \node[flowstep, stageteal, below=44mm of firstline] (cohort)
    {Final cohort\\\cfn{37,531}\\\cfsub{models trained, tuned and tested on these patients}};
  \node[splitbox, below=10mm of cohort] (tune)
    {Tuning\\\cfn{1,980}\\\cfsub{index in 2022}};
  \node[splitbox, left=7mm of tune] (train)
    {Training\\\cfn{29,105}\\\cfsub{index through 2021}};
  \node[splitbox, right=7mm of tune] (test)
    {Test (held out)\\\cfn{6,446}\\\cfsub{index from 2023}};

  % Main flow and split
  \foreach \a/\b in {pre/index, index/classified,
                     classified/firstline, firstline/cohort}
    \draw[flowblue] (\a.south) -- (\b.north);
  \coordinate (fork) at ($(cohort.south)+(0,-6mm)$);
  \draw[line width=0.75pt, draw=OperaTeal] (cohort.south) -- (fork);
  \draw[line width=0.75pt, draw=OperaTeal] (train.north |- fork) -- (test.north |- fork);
  \foreach \s in {train, tune, test}
    \draw[flowteal] (\s.north |- fork) -- (\s.north);

  % Exclusions within a band leave the line midway between two steps.
  \foreach \a/\b/\name/\body in {
      index/classified/x2/{\textbf{Excluded, \textit{n}\,=\,1,069}\\Diagnosis unresolved after both passes},
      classified/firstline/x3/{\textbf{Excluded, \textit{n}\,=\,9}\\IgM MGUS, untreated by definition}} {
    \coordinate (m\name) at ($(\a.south)!0.5!(\b.north)$);
    \node[excl, anchor=west] (\name) at ($(m\name)+(35mm,0)$) {\body};
    \draw[exclarrow] (m\name) -- (\name.west);
  }

  % Bands behind everything, one width; eyebrow and heading at the top left.
  \coordinate (L) at ($(train.west)+(-1mm,0)$);
  \coordinate (R) at ($(x2.east)+(1mm,0)$);
  \begin{scope}[on background layer]
    \node[band, fit=(pre)(L |- {$(pre.north)+(0,9mm)$})(R |- pre.south)] (bpre) {};
    \node[band, fit=(index)(firstline)(L |- {$(index.north)+(0,9mm)$})(R |- firstline.south)] (basc) {};
    \node[band, fit=(cohort)(train)(test)(L |- {$(cohort.north)+(0,9mm)$})(R |- train.south)] (bsup) {};
  \end{scope}
  % Exclusions between bands sit in the middle of the white gap.
  \foreach \up/\down/\name/\body in {
      bpre/basc/x1/{\textbf{Excluded, \textit{n}\,=\,40,033}\\No qualifying first-line treatment},
      basc/bsup/x4/{\textbf{Excluded, \textit{n}\,=\,26}\\Died (\textit{n}\,=\,17) or follow-up ended
        (\textit{n}\,=\,9) before treatment}} {
    \coordinate (m\name) at ($(\up.south)!0.5!(\down.north)$);
    \coordinate (m\name) at (pre.south |- m\name);
    \node[excl, anchor=west] (\name) at ($(m\name)+(35mm,0)$) {\body};
    \draw[exclarrow] (m\name) -- (\name.west);
  }
  \foreach \b/\colour/\title in {bpre/OperaIndigo/Pretraining population,
                                 basc/OperaInkSecondary/First-line ascertainment,
                                 bsup/OperaTeal/Prediction cohort}
    \node[heading, font=\sffamily\fontsize{11}{13}\selectfont\bfseries, text=\colour,
          anchor=north west] at ($(\b.north west)+(3.4mm,-3.0mm)$) {\title};
\end{tikzpicture}
